# Optional Luma configuration overrides
#
# The .env file is used to override the default settings. Save a copy of this file as
# .env and edit it as needed. Changes require a restart to take effect.
#
# Where .env should be stored:
#   Portable download (Windows, Linux)   the folder holding the luma program
#   macOS app                            ~/Library/Application Support/Luma
#   Source checkout                      the top folder of the checkout
#   Docker                               the folder holding docker-compose.yml
#
# Settings -> Configuration shows the exact path (whether or not the file exists yet), every effective
# value, and where each value came from.
#
# Not every setting applies to every installation:
#   [all]       applies to every installation
#   [download]  a downloaded build: the portable download or the macOS app
#   [source]    running from a source checkout with the start scripts
#   [docker]    running with docker-compose
#

# ------------------------------------------------------------------------------
# 1. Where the data lives
# ------------------------------------------------------------------------------

# DATA_DIR [all]
# The folder holding both databases (luma-data.db, luma-control.db) and the documents folder. Pointing a
# second copy of Luma at a different folder is how two installations share one machine.
# Default:
#   Portable download   the folder holding the luma program (a relative path is taken from there too)
#   macOS app           ~/Library/Application Support/Luma
#   Source checkout     the folder the backend is started from
#   Docker              /data inside the container, set by docker-compose.yml (not changeable here)
# DATA_DIR=

# ------------------------------------------------------------------------------
# 2. Ports
# ------------------------------------------------------------------------------
#
# The browser always connects to the frontend port. A downloaded build is one program on that one port,
# and so is the application under Docker, where it is the port the proxy beside it passes requests to.
# Running from source adds a separate frontend server, which passes API requests on to the backend port
# behind it.

# FRONTEND_PORT [all]
# The port the browser connects to. If it is busy, a downloaded build starts on a free port and reports
# which: in the terminal, or in luma-desktop.log beside the data for the macOS app. The start scripts
# refuse to start instead, naming the busy port, and so does a downloaded build in server mode or the
# application under Docker.
# Default: 8010
# FRONTEND_PORT=8010

# BACKEND_HOST [source]
# Address the backend listens on. This computer only by default. 0.0.0.0 makes it reachable from other
# devices on the network, where it serves the built interface as well, over plain HTTP: from another
# device there are no passkeys, and a data source can only be connected from this computer.
# Default: 127.0.0.1
# BACKEND_HOST=0.0.0.0

# BACKEND_PORT [source]
# The port the backend listens on behind the frontend server. A downloaded build does not use it.
# Default: 8011
# BACKEND_PORT=8011

# FRONTEND_HOST [download] [source]
# Address the browser connects to: the development frontend from source, and a downloaded build in
# server mode. This computer only by default. 0.0.0.0 makes it reachable from other devices on the
# network over plain HTTP, with the limits described under BACKEND_HOST; a downloaded build in server
# mode is otherwise meant to sit behind a reverse proxy that provides HTTPS. Outside server mode a
# downloaded build always accepts connections from this computer only. Default: localhost
# FRONTEND_HOST=0.0.0.0

# ------------------------------------------------------------------------------
# 3. Sessions and deployment
# ------------------------------------------------------------------------------

# SESSION_COOKIE_NAME [all]
# Keeps browser sessions apart when two installations are used from the same browser.
# Default: derived from the control database path on first run, then stored
# SESSION_COOKIE_NAME=

# SESSION_SECRET [all]
# Signs session cookies. Generated on first run and stored in the control database; set here only to pin
# a specific value, for example to keep sessions valid across a rebuilt control database. Changing it
# signs everyone out.
# SESSION_SECRET=

# SERVER_MODE [download]
# Runs a downloaded build as a server for other devices to reach: no browser is opened, the port is
# fixed, and the app cannot be shut down from inside it. Other devices reach it through a reverse proxy
# on the same machine that provides HTTPS, or directly over plain HTTP when FRONTEND_HOST is widened.
# Starting the program with --server or --no-server overrides this. Default: false
# SERVER_MODE=true

# MDNS_NAME [download] [docker]
# The name a server-mode installation answers to on the local network, without the .local ending, so
# other devices reach it at <name>.local. A reverse proxy in front has to be configured with the same
# name. Two installations on one network need different names: one already in use refuses to start.
# Default: luma
# MDNS_NAME=luma

# SESSION_COOKIE_SECURE [all]
# Sends the session cookie only over HTTPS. For a server installation behind a TLS-terminating reverse
# proxy. A downloaded build outside server mode serves plain HTTP to this computer only and must leave
# it off. Default: false
# SESSION_COOKIE_SECURE=false

# CORS_ORIGINS [source] [docker]
# Other web origins whose pages may call this API from a browser, with cookies attached. Needed only for
# a deployment that serves the interface and the API from different origins. Default: empty
# CORS_ORIGINS=https://app.example.invalid

# APP_NAME [source] [docker]
# Display name, for white-labelling. The interface takes its name from VITE_APP_NAME when it is built,
# so this is only meaningful alongside a build of the interface. Default: Luma
# APP_NAME=Luma

# ------------------------------------------------------------------------------
# 4. Launcher and developer options
# ------------------------------------------------------------------------------

# LUMA_DEVELOPER_OPTIONS [all]
# Shows developer-only options in Settings, including destructive reset tools. Not needed for normal use.
# LUMA_DEVELOPER_OPTIONS=true

# LUMA_USE_LOG_FILES [source]
# Saves the start scripts' backend and frontend terminal output beside the scripts, overwritten on each
# start. The server's own luma.log, beside the data, is written either way. Default: false
# LUMA_USE_LOG_FILES=false

# LUMA_BACKEND_READY_TIMEOUT [source]
# Seconds the start scripts wait for the backend before starting the frontend anyway. Worth raising if
# the first run's dependency installation is slow. Default: 180
# LUMA_BACKEND_READY_TIMEOUT=180

# VITE_API_PROXY_TARGET [source]
# Where the development frontend sends API requests. Derived from BACKEND_PORT when unset.
# VITE_API_PROXY_TARGET=http://localhost:8011

# LUMA_ENV_FILE [source] cannot be set in this file, because it chooses which file is read. Set in the
# environment before starting the backend, it points the backend (only) at a different override file;
# the start scripts and the development frontend still read this one for their ports. The browser test
# harness uses it to run against throwaway databases.
